class A
var
begin
  method fun():
  var
  begin
    print!("a, fun");
    print!(this);
  end;
end;

class B<T>
var
begin
  method build():
  var
  begin
    return make!<T>();
  end;
end;

method main(): int
var
  a, b;
begin
  b = make!<B<A>>();
  a = b.build();

  print!(type!(a));
  a.fun();

  return cast!<int>(0);
end;
